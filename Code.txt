# Install-SqlClient.ps1 - PowerShell 5.1, run as Administrator on each agent
[Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12

$InstallDir = 'C:\Program Files\SqlClient'
$LibDir     = Join-Path $InstallDir 'lib'
$PkgDir     = Join-Path $env:TEMP 'SqlClient-packages'
$NuGetExe   = Join-Path $env:TEMP 'nuget.exe'
$Source     = 'https://api.nuget.org/v3/index.json'   # swap for your Artifactory NuGet remote
$Framework  = 'net462'

if (Test-Path $PkgDir) { Remove-Item $PkgDir -Recurse -Force }
if (Test-Path $LibDir) { Remove-Item $LibDir -Recurse -Force }
New-Item -ItemType Directory -Path $PkgDir, $LibDir -Force | Out-Null

if (-not (Test-Path $NuGetExe)) {
    Invoke-WebRequest -Uri 'https://dist.nuget.org/win-x86-commandline/latest/nuget.exe' -OutFile $NuGetExe -UseBasicParsing
}

& $NuGetExe install Microsoft.Data.SqlClient -OutputDirectory $PkgDir -Framework $Framework -Source $Source -NonInteractive
if ($LASTEXITCODE -ne 0) { throw "nuget install failed ($LASTEXITCODE)" }

# Highest version of each DLL wins
$packages = Get-ChildItem $PkgDir -Directory | ForEach-Object {
    $ver = [version]'0.0'
    if ($_.Name -match '^(?<id>.+?)\.(?<ver>\d+(\.\d+){1,3})(-.*)?$') { $ver = [version]$Matches['ver'] }
    [pscustomobject]@{ Dir = $_; Version = $ver }
} | Sort-Object Version

$Preferred = 'net462','net461','net46','net45','netstandard2.0'
foreach ($pkg in $packages) {
    $libRoot = Join-Path $pkg.Dir.FullName 'lib'
    if (-not (Test-Path $libRoot)) { continue }
    foreach ($fw in $Preferred) {
        $fwPath = Join-Path $libRoot $fw
        if (Test-Path $fwPath) {
            Get-ChildItem $fwPath -Filter '*.dll' | ForEach-Object {
                $dest = Join-Path $LibDir $_.Name
                $copy = $true
                if (Test-Path $dest) {
                    $copy = [Reflection.AssemblyName]::GetAssemblyName($_.FullName).Version -ge
                            [Reflection.AssemblyName]::GetAssemblyName($dest).Version
                }
                if ($copy) { Copy-Item $_.FullName $dest -Force }
            }
            break
        }
    }
}

# Native SNI DLLs (both architectures; loader uses the right one)
foreach ($arch in 'x64','x86') {
    Get-ChildItem $PkgDir -Recurse -Filter "Microsoft.Data.SqlClient.SNI.$arch.dll" |
        Sort-Object FullName -Descending | Select-Object -First 1 |
        Copy-Item -Destination $LibDir -Force
}

# Write the loader beside the DLLs
$loader = @'
# Load-SqlClient.ps1 - dot-source this: . "$env:SQLCLIENT_HOME\Load-SqlClient.ps1"
$SqlClientLib = Join-Path $PSScriptRoot 'lib'

if (-not ('SqlClientResolver' -as [type])) {
    Add-Type -TypeDefinition @"
using System;
using System.IO;
using System.Reflection;

public static class SqlClientResolver
{
    private static string _dir;

    public static void Register(string dir)
    {
        _dir = dir;
        AppDomain.CurrentDomain.AssemblyResolve += OnResolve;
    }

    private static Assembly OnResolve(object sender, ResolveEventArgs args)
    {
        string name = new AssemblyName(args.Name).Name;
        string path = Path.Combine(_dir, name + ".dll");
        return File.Exists(path) ? Assembly.LoadFrom(path) : null;
    }
}
"@
    [SqlClientResolver]::Register($SqlClientLib)
}

[void][Reflection.Assembly]::LoadFrom((Join-Path $SqlClientLib 'Microsoft.Data.SqlClient.dll'))
'@
Set-Content -Path (Join-Path $InstallDir 'Load-SqlClient.ps1') -Value $loader -Encoding UTF8

# Machine env var so pipelines can find it
[Environment]::SetEnvironmentVariable('SQLCLIENT_HOME', $InstallDir, 'Machine')

Remove-Item $PkgDir -Recurse -Force

# Restart agent service(s) so they pick up the new environment variable
Get-Service -Name 'vstsagent*', 'azpipelinesagent*' -ErrorAction SilentlyContinue | Restart-Service

$v = [Reflection.AssemblyName]::GetAssemblyName((Join-Path $LibDir 'Microsoft.Data.SqlClient.dll')).Version
Write-Host "Microsoft.Data.SqlClient $v installed to $InstallDir"
