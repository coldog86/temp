# stop in order
sudo systemctl stop xray
sudo systemctl stop postgresql        # use the exact service name from above

# Xray data
sudo mkdir -p /data/xray-updates/xray
sudo rsync -aHAX /var/opt/jfrog/xray/ /data/xray-updates/xray/
sudo mv /var/opt/jfrog/xray /var/opt/jfrog/xray.old
sudo mkdir /var/opt/jfrog/xray

# Postgres data
sudo mkdir -p /data/xray-updates/pgsql
sudo rsync -aHAX /var/lib/pgsql/ /data/xray-updates/pgsql/
sudo mv /var/lib/pgsql /var/lib/pgsql.old
sudo mkdir /var/lib/pgsql

# bind mounts (after the sdb mount so boot order is right)
echo '/data/xray-updates/xray  /var/opt/jfrog/xray  none  bind  0 0' | sudo tee -a /etc/fstab
echo '/data/xray-updates/pgsql /var/lib/pgsql       none  bind  0 0' | sudo tee -a /etc/fstab
sudo systemctl daemon-reload
sudo mount /var/opt/jfrog/xray
sudo mount /var/lib/pgsql

# ownership and SELinux labels (policy is path-based, so bind mounts at the original paths get the right contexts)
sudo chown xray:xray /var/opt/jfrog/xray
sudo chown postgres:postgres /var/lib/pgsql
sudo restorecon -Rv /var/opt/jfrog/xray /var/lib/pgsql

# start in order
sudo systemctl start postgresql
sudo systemctl start xray



ps aux | grep [r]sync
sudo du -sh /data/xray-updates/xray; sleep 30; sudo du -sh /data/xray-updates/xray
