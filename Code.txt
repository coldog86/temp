# Generate a self-signed cert valid for localhost
$cert = New-SelfSignedCertificate -DnsName "localhost" -CertStoreLocation "cert:\LocalMachine\My" -NotAfter (Get-Date).AddYears(2)
$thumbprint = $cert.Thumbprint
Write-Host "Thumbprint: $thumbprint"

# Bind it to port 443 for the CCM site in IIS
Import-Module WebAdministration
New-WebBinding -Name "ChocolateyCentralManagement" -Protocol https -Port 443 -IPAddress "*"

# Assign the cert to that binding via netsh (IIS's binding cmdlets don't set the cert itself)
$appId = "{00000000-0000-0000-0000-000000000000}"
netsh http add sslcert ipport=0.0.0.0:443 certhash=$thumbprint appid=$appId
