# Stop everything first
Stop-Website -Name ChocolateyCentralManagement -ErrorAction SilentlyContinue
Stop-Service chocolatey-central-management -ErrorAction SilentlyContinue

# Uninstall the packages
choco uninstall chocolatey-management-web -y --force
choco uninstall chocolatey-management-service -y --force
choco uninstall chocolatey-management-database -y --force

# Drop the database entirely rather than trying to patch a broken schema.
# SET SINGLE_USER forces any lingering connections closed before the drop.
sqlcmd -S "localhost\SQLEXPRESS" -E -Q "IF EXISTS (SELECT 1 FROM sys.databases WHERE name = 'ChocolateyManagement') BEGIN ALTER DATABASE ChocolateyManagement SET SINGLE_USER WITH ROLLBACK IMMEDIATE; DROP DATABASE ChocolateyManagement; END"

# Clean up leftover folders choco uninstall doesn't always fully remove
Remove-Item "C:\tools\chocolatey-management-web" -Recurse -Force -ErrorAction SilentlyContinue
Get-ChildItem "C:\ProgramData\chocolatey\lib" -Filter "chocolatey-management-*" -Directory | Remove-Item -Recurse -Force -ErrorAction SilentlyContinue

# Remove the IIS site/app pool if choco uninstall left them behind
Import-Module WebAdministration -ErrorAction SilentlyContinue
Remove-Website -Name "ChocolateyCentralManagement" -ErrorAction SilentlyContinue
Remove-WebAppPool -Name "ChocolateyCentralManagement" -ErrorAction SilentlyContinue

$hostName = [System.Net.Dns]::GetHostName()
$domainName = [System.Net.NetworkInformation.IPGlobalProperties]::GetIPGlobalProperties().DomainName
$fqdnUser = "$hostName\ChocolateyLocalAdmin"

$grantSql = @"
IF NOT EXISTS (SELECT 1 FROM sys.server_principals WHERE name = '$fqdnUser')
    CREATE LOGIN [$fqdnUser] FROM WINDOWS;
IF NOT EXISTS (SELECT 1 FROM sys.server_principals WHERE name = 'IIS APPPOOL\ChocolateyCentralManagement')
    CREATE LOGIN [IIS APPPOOL\ChocolateyCentralManagement] FROM WINDOWS;

USE ChocolateyManagement;

IF NOT EXISTS (SELECT 1 FROM sys.database_principals WHERE name = '$fqdnUser')
    CREATE USER [$fqdnUser] FOR LOGIN [$fqdnUser];
IF NOT EXISTS (SELECT 1 FROM sys.database_principals WHERE name = 'IIS APPPOOL\ChocolateyCentralManagement')
    CREATE USER [IIS APPPOOL\ChocolateyCentralManagement] FOR LOGIN [IIS APPPOOL\ChocolateyCentralManagement];

ALTER ROLE db_datareader ADD MEMBER [$fqdnUser];
ALTER ROLE db_datawriter ADD MEMBER [$fqdnUser];
ALTER ROLE db_datareader ADD MEMBER [IIS APPPOOL\ChocolateyCentralManagement];
ALTER ROLE db_datawriter ADD MEMBER [IIS APPPOOL\ChocolateyCentralManagement];
"@

sqlcmd -S "localhost\SQLEXPRESS" -E -Q $grantSql
