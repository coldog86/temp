# Invoke-SnowRequest
function Invoke-SnowRequest(){
    param(
        [Parameter(Mandatory = $true)][string] $Uri,
        [Parameter(Mandatory = $true)] $SnowBearerToken,
        [Parameter(Mandatory = $true)] $JsonBody
    )
    try {
        $response = Invoke-RestMethod -Uri $Uri -Method POST `
            -UseBasicParsing `
            -Headers @{
                Accept        = "application/json";
                Authorization = "Bearer $($SnowBearerToken)"
            } `
            -ContentType "application/json" `
            -Body $JsonBody
    }
    catch {
        Write-Host "##[error] Message: $($_.Exception.Message)"

        if ($_.Exception.Response) {
            $statusCode = [int]$_.Exception.Response.StatusCode
            Write-Host "##[error] HTTP Status: $statusCode"

            $stream = $_.Exception.Response.GetResponseStream()
            $stream.Position = 0
            $reader = New-Object System.IO.StreamReader($stream)
            $rawBody = $reader.ReadToEnd()
            $reader.Close()

            Write-Host "##[error] Raw body: $rawBody"

            try {
                $errorJson = $rawBody | ConvertFrom-Json
                Write-Host "##[error] Parsed error: $($errorJson.error | ConvertTo-Json -Depth 5)"
                return $errorJson
            }
            catch {
                Write-Host "##[error] Body was not valid JSON."
            }
        }
        else {
            Write-Host "##[error] No Response object on the exception — likely DNS/TLS/connectivity failure, not an HTTP error."
        }
    }
    return $response
}

# Get-FormVariables 
function Get-FormVariables(){
    param(
        [Parameter(Mandatory = $true)] $SnowBearerToken,
        [Parameter(Mandatory = $true)] $FormSysId,
        [Parameter(Mandatory = $false)][switch] $DevCall
    )
    Write-Host "##[command] ##[FUNCTION] Get-FormVariables -FormSysId $formSysId -SnowBearerToken $snowBearerToken -DevCall $devCall"

    if($devCall){
        $instance = "desedev.service-now.com" 
    } else {
        $instance = "dese.service-now.com"
    }

    $vars = @()
    $varsUri = "https://$instance/api/now/table/item_option_new?sysparm_query=cat_item=$formSysId"

    $directVars = Invoke-RestMethod `
        -Uri $varsUri `
        -Method GET `
        -Headers @{
            Accept = "application/json"
            Authorization = "Bearer $snowBearerToken"
        }
    foreach ($directVar in $directVars.result){
        $vars += $directVar | Select name, question_text, type, mandatory, reference, sys_id, variable_set
    }

    $setUri = "https://$instance/api/now/table/io_set_item?sysparm_query=sc_cat_item=$formSysId"
    $sets = Invoke-RestMethod `
        -Uri $setUri `
        -Headers @{
            Accept = "application/json"
            Authorization = "Bearer $snowBearerToken"
        }


    foreach ($set in $sets.result) {
        $setId = $set.variable_set.value

        $setVarsUri = "https://$instance/api/now/table/item_option_new?sysparm_query=variable_set=$setId&sysparm_fields=name,question_text,type,mandatory,reference,sys_id,variable_set"

        $setVars = Invoke-RestMethod `
            -Uri $setVarsUri `
            -Headers @{
                Accept = "application/json"
                Authorization = "Bearer $snowBearerToken"
            }

        foreach ($setVar in $setVars.result){
            $vars += $setVar | Select name, question_text, type, mandatory, reference, sys_id, variable_set
        }
    }

    return $vars
}

            work_end = $r.work_end

    else{

    }

# Get-CmdbOperationalVms
function Get-CmdbOperationalVms(){
    param(
        [Parameter(Mandatory = $true)] $SnowBearerToken,
        [Parameter(Mandatory = $false)][int] $limit = 10000,
        [Parameter(Mandatory = $false)][switch] $DevCall
    )
    Write-Host "##[command] ##[FUNCTION] Get-CmdbOperationalVms -SnowBearerToken $snowBearerToken -Limit $limit -DevCall $devCall"
    
    if($devCall){
        $vms = Get-CmdbVm -SnowBearerToken $snowBearerToken -Limit $limit -DevCall
    } else {
        $vms = Get-CmdbVm -SnowBearerToken $snowBearerToken -Limit $limit
    }

    
    
    $operationalVms = $vms | Where-Object {
        $_.life_cycle_stage_status.value -eq 'In Use' -and
        $_.life_cycle_stage.value -eq 'Operational'
    }

    Write-Host "Found $($operationalVms.count) operational servers"
    return $operationalVms
}

# Get-SnowToken
function Get-SnowToken(){
    param(
        [Parameter(Mandatory = $true)][string] $SnowClientID,
        [Parameter(Mandatory = $true)] $SnowAPIKey,
        [Parameter(Mandatory = $true)][string] $SnowApiUser,
        [Parameter(Mandatory = $true)] $SnowApiUserPassword
    )

    $snowBearerToken = Invoke-RestMethod `
        -Method POST `
        -Uri 'https://login.microsoftonline.com/dd0cfd15-4558-4b12-8bad-ea26984fc417/oauth2/v2.0/token' `
        -ContentType 'application/x-www-form-urlencoded' `
        -UseBasicParsing `
        -body @{
            grant_type = "password";
            scope = 'openid email profile';
            response_type = "id_token";
            response_mode = "form_post";
            client_id = $snowClientID ;
            client_secret = $snowAPIKey;
            username = $snowApiUser;
            password = $snowApiUserPassword;
        } 

    return $snowBearerToken.id_token

#  Get-Operational_Cmdb_Vms_In_Departed_Ou
function Get-Operational_Cmdb_Vms_In_Departed_Ou (){    
    param(
        [Parameter(Mandatory = $true)] $SnowBearerToken,
        [Parameter(Mandatory = $true)] $VraBearerToken,
        [Parameter(Mandatory = $true)] $InfobloxCredential,
        [Parameter(Mandatory = $false)] $Domain
    )
    Write-Host "##[command] ##[FUNCTION] Get-Operational_Cmdb_Vms_In_Departed_Ou -SnowBearerToken $snowBearerToken -VraBearerToken $vraBearerToken -InfobloxCredential $infobloxCredential -Domain $domain"

    $domainobjs = Get-DomainObjs -Domain $domain
    Write-Host "Getting operational VMs from CMDB" -ForegroundColor Cyan
    $operationalSnowVms = Get-CmdbOperationalVms -SnowBearerToken $snowBearerToken
    $operational_Cmdb_Vms_In_Departed_Ou = @()

    foreach($obj in $domainobjs){
        $departedVms = Get-ADComputer -SearchBase $obj.departedOU -SearchScope Subtree -Filter * -Server $obj.domain -Properties DistinguishedName,Enabled,DNSHostName,LastLogonDate,WhenCreated,OperatingSystem,IPv4Address
        Write-Host "There are $($departedVms.Count) VMs in the departed OU for $($obj.domain)" -ForegroundColor Cyan        
        
        $operationalDomainVms = @()
        foreach($vm in $operationalSnowVms){
            if($vm.os_domain -eq $obj.domain){
                $operationalDomainVms += $vm
            }
        }
        Write-Host "There are $($operationalDomainVms.count) 'operational' VMs in $($obj.domain) according to CMDB"

        $departedOperationalVms = @()
        foreach($vm in $operationalDomainVms){
            if($departedVms.name -contains $vm.name){
                $departedOperationalVms += $vm
            }
        }

        foreach($vm in $departedOperationalVms){
            $vmObj = Populate-VmObj -VmName $vm.name -Vmdomain $obj.domain -SnowBearerToken $snowBearerToken -VraBearerToken $vraBearerToken -InfobloxCredential $infobloxCredential  
            $operational_Cmdb_Vms_In_Departed_Ou += $vmObj
            $count++
        }
        Write-Host "There are $($count) operational VMs according to SNOW in the dearted OU for $($obj.domain)" -ForegroundColor Cyan
    }
    return $operational_Cmdb_Vms_In_Departed_Ou
}

# Get-DomainObjs
function Get-DomainObjs(){
    param(
        [Parameter(Mandatory = $false)] $Domain
    )
    Write-Host "##[command] ##[FUNCTION] Get-DomainObjs -Domain $domain"
    
    $domainobj = @()
    $domainobj += [pscustomobject]@{ 
        domain = "application.enet"
        departedOU = 'OU=Computers Departed,DC=application,DC=enet'
    }
    $domainobj += [pscustomobject]@{ 
        domain = "construction.enet"
        departedOU = 'OU=Computers Departed,DC=construction,DC=enet'
    }
    $domainobj += [pscustomobject]@{ 
        domain = "dev.construction.enet"
        departedOU = 'OU=Computers Departed,DC=dev,DC=construction,DC=enet'
    }
    $domainobj += [pscustomobject]@{ 
        domain = "pprod.construction.enet"
        departedOU = 'OU=Computers Departed,DC=pprod,DC=construction,DC=enet'
    }

    if($domain -ne $null){
        if($domain -eq 'prod'){
            $domainobj = [pscustomobject]@{ 
                domain = "application.enet"
                departedOU = 'OU=Computers Departed,DC=application,DC=enet'
            }
        }
        if($domain -eq 'construction'){
            $domainobj = [pscustomobject]@{ 
                domain = "construction.enet"
                departedOU = 'OU=Computers Departed,DC=construction,DC=enet'
            }
        }
        if($domain -eq 'dev'){
            $domainobj = [pscustomobject]@{ 
                domain = "dev.construction.enet"
                departedOU = 'OU=Computers Departed,DC=dev,DC=construction,DC=enet'
            }
        }
        if($domain -eq 'pprod'){
            $domainobj = [pscustomobject]@{ 
                domain = "pprod.construction.enet"
                departedOU = 'OU=Computers Departed,DC=pprod,DC=construction,DC=enet'
            }
        }
    }
    return $domainobj
}

# Populate-VmObj
function Populate-VmObj (){
    param(
        [Parameter(Mandatory = $true)] $VmName,
        [Parameter(Mandatory = $true)] $VmDomain,
        [Parameter(Mandatory = $true)] $SnowBearerToken,
        [Parameter(Mandatory = $true)] $VraBearerToken,
        [Parameter(Mandatory = $true)] $InfobloxCredential
    )
    Write-Host "##[command] ##[FUNCTION] Populate-VmObj -SnowBearerToken $snowBearerToken -VraBearerToken $vraBearerToken -InfobloxCredential $infobloxCredential -VmName $vmName -VmDomain $vmDomain"
    
    
    $vmObj = get-VmObject
    $adobj = Get-ADComputer $vmName -server $vmDomain -Properties DistinguishedName,Enabled,DNSHostName,LastLogonDate,WhenCreated,OperatingSystem,IPv4Address,PrimaryGroup
    $decomRequestObj = Get-CmdbVmDecommisionRequest -VmName $vmName -SnowBearerToken $snowBearerToken
    $infobloxRecord = Get-InfobloxRecord -InfobloxCredential $infobloxCredential -RecordType a -RecordName $adobj.DNSHostName -Silent
    $vraMachine = Get-VraMachine -BearerToken $vraBearerToken -VmName $vmName
    $cmdbVm = Get-CmdbVm -VmName $vmName -SnowBearerToken $snowBearerToken

    # AD Attributes
    $vmObj.Name = $adobj.name
    $vmObj.PowerState = $vraMachine.powerState
    $vmObj.EnabledInAD = $adobj.Enabled
    $vmObj.DistinguishedName = $adobj.DistinguishedName
    $vmObj.DNSHostName = $adobj.DNSHostName
    $vmObj.LastLogonDate = $adobj.LastLogonDate
    $vmObj.WhenCreated = $adobj.WhenCreated
    $vmObj.OperatingSystem = $adobj.OperatingSystem
    $vmObj.IPv4AddressInAD = $adobj.IPv4Address
    $vmObj.PrimaryGroup = $adobj.PrimaryGroup
                
    # Infoblox
    $vmObj.InfoBloxRecordName = $infobloxRecord.name
    $vmObj.InfoBloxIPaddress= $infobloxRecord.ipv4addrs.ipv4addr
                
    # Decom Request
    $vmObj.DecomRequest_task_effective_number = $decomRequestObj.task_effective_number
    $vmObj.DecomRequest_approval = $decomRequestObj.approval 
    $vmObj.DecomRequest_short_description = $decomRequestObj.short_description 
    $vmObj.DecomRequest_close_code = $decomRequestObj.close_code
    $vmObj.DecomRequest_closed_at = $decomRequestObj.closed_at 
    $vmObj.DecomRequest_closed_by = $decomRequestObj.closed_by
    $vmObj.DecomRequest_phase_state = $decomRequestObj.phase_state
    $vmObj.DecomRequest_opened_at = $decomRequestObj.opened_at
    $vmObj.DecomRequest_work_end = $decomRequestObj.work_end
    $vmObj.DecomRequest_opened_by = $decomRequestObj.opened_by 
    $vmObj.DecomRequest_assigned_to = $decomRequestObj.assigned_to	
                        
    # CMDB
    $vmObj.CMDB_name = $cmdbVm.name
    $vmObj.CMDB_ip_address = $cmdbVm.ip_address	
    $vmObj.CMDB_fqdn = $cmdbVm.fqdn	
    $vmObj.CMDB_last_discovered	= $cmdbVm.last_discovered	
    $vmObj.CMDB_os = $cmdbVm.os
    $vmObj.CMDB_os_version = $cmdbVm.os_version
    $vmObj.CMDB_life_cycle_stage = $cmdbVm.life_cycle_stage.value
    $vmObj.CMDB_life_cycle_stage_status	= $cmdbVm.life_cycle_stage_status.value
    $vmObj.CMDB_managed_by = if ($null -ne $cmdbVm.managed_by.link){ (Get-SnowQuery -Uri $cmdbVm.managed_by.link -SnowBearerToken $snowBearerToken).result.name	 } else { $null }
    $vmObj.CMDB_u_tags = $cmdbVm.u_tags	
    $vmObj.CMDB_u_tags_ProjectCode = if($null -ne $cmdbVm.u_tags){ ($cmdbVm.u_tags | Convertfrom-Json).'Project Code/Cost Centre Code' } else { $null }
    $vmObj.CMDB_u_tags_Environment = if($null -ne $cmdbVm.u_tags){ ($cmdbVm.u_tags | Convertfrom-Json).Environment } else { $null }
    $vmObj.CMDB_u_tags_OperatingSystem = if($null -ne $cmdbVm.u_tags){ ($cmdbVm.u_tags | Convertfrom-Json).OperatingSystem } else { $null }
    $vmObj.CMDB_u_tags_ProjectName = if($null -ne $cmdbVm.u_tags){ ($cmdbVm.u_tags | Convertfrom-Json).'Project Name' } else { $null }
    $vmObj.CMDB_u_tags_Agency = if($null -ne $cmdbVm.u_tags){ ($cmdbVm.u_tags | Convertfrom-Json).Agency } else { $null }
    $vmObj.CMDB_company = if ($null -ne $cmdbVm.company.link){ (Get-SnowQuery -Uri $cmdbVm.company.link -SnowBearerToken $snowBearerToken).result.name  } else { $null }
    $vmObj.CMDB_u_function = $cmdbVm.u_function
    $vmObj.CMDB_comments = $cmdbVm.comments	
    $vmObj.CMDB_u_purpose = $cmdbVm.u_purpose	
    $vmObj.CMDB_short_description = $cmdbVm.short_description
    $vmObj.CMDB_sys_updated_on = $cmdbVm.sys_updated_on

    # Timestamp
    $vmObj.TimeStamp = Get-Date


    return $vmObj               	
}
