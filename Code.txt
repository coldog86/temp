function Get-VraVmResources {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)] [object[]] $Resources,
        [Parameter(Mandatory)] [string]   $VmName
    )

    $machines = @($Resources | Where-Object { $_.type -eq 'Cloud.Machine' })
    $machine  = $machines | Where-Object { $_.properties.resourceName -eq $VmName }

    if (-not $machine) {
        throw "No Cloud.Machine with resourceName '$VmName' found in deployment (search may have matched a similar name)."
    }
    if (@($machine).Count -gt 1) {
        throw "Multiple Cloud.Machine resources named '$VmName' found - refusing to continue."
    }

    # Volumes are matched by the vSphere moref they are attached to, not by index
    $volumes = @($Resources | Where-Object {
        $_.type -eq 'Cloud.Volume' -and $_.properties.vm -eq $machine.properties.moref
    })

    [pscustomobject]@{
        Machine           = $machine
        Volumes           = $volumes
        TotalMachineCount = $machines.Count
    }
}

function Get-VraResourceActions {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)] [string] $BaseUrl,
        [Parameter(Mandatory)] [string] $BearerToken,
        [Parameter(Mandatory)] [string] $DeploymentId,
        [Parameter(Mandatory)] [string] $ResourceId
    )

    $headers = @{ Authorization = "Bearer $BearerToken" }
    $uri     = "$BaseUrl/deployment/api/deployments/$DeploymentId/resources/$ResourceId/actions"
    Write-Host "##[debug] URI = $uri"

    @(Invoke-RestMethod -Method Get -Uri $uri -Headers $headers)
}

function Invoke-VraResourceAction {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)] [string] $BaseUrl,
        [Parameter(Mandatory)] [string] $BearerToken,
        [Parameter(Mandatory)] [string] $DeploymentId,
        [Parameter(Mandatory)] [string] $ResourceId,
        [Parameter(Mandatory)] [string] $ActionId,
        [string] $Reason = 'Decommission'
    )

    $headers = @{ Authorization = "Bearer $BearerToken" }
    $uri     = "$BaseUrl/deployment/api/deployments/$DeploymentId/resources/$ResourceId/requests"
    $body    = @{ actionId = $ActionId; inputs = @{}; reason = $Reason } | ConvertTo-Json -Depth 5

    Write-Host "##[command] POST $uri (actionId = $ActionId)"
    Invoke-RestMethod -Method Post -Uri $uri -Headers $headers -Body $body -ContentType 'application/json'
}

function Remove-VraDeploymentWhole {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)] [string] $BaseUrl,
        [Parameter(Mandatory)] [string] $BearerToken,
        [Parameter(Mandatory)] [string] $DeploymentId,
        [string] $Reason = 'Decommission'
    )

    $headers = @{ Authorization = "Bearer $BearerToken" }
    $uri     = "$BaseUrl/deployment/api/deployments/$DeploymentId/requests"
    $body    = @{ actionId = 'Deployment.Delete'; inputs = @{}; reason = $Reason } | ConvertTo-Json -Depth 5

    Write-Host "##[command] POST $uri (actionId = Deployment.Delete)"
    Invoke-RestMethod -Method Post -Uri $uri -Headers $headers -Body $body -ContentType 'application/json'
}

function Wait-VraRequest {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory)] [string] $BaseUrl,
        [Parameter(Mandatory)] [string] $BearerToken,
        [Parameter(Mandatory)] [string] $RequestId,
        [int] $TimeoutMinutes = 30,
        [int] $PollSeconds    = 15
    )

    $headers  = @{ Authorization = "Bearer $BearerToken" }
    $uri      = "$BaseUrl/deployment/api/requests/$RequestId"
    $deadline = (Get-Date).AddMinutes($TimeoutMinutes)

    do {
        Start-Sleep -Seconds $PollSeconds
        $req = Invoke-RestMethod -Method Get -Uri $uri -Headers $headers
        Write-Host "##[debug] Request $RequestId status = $($req.status)"

        switch ($req.status) {
            'SUCCESSFUL' { return $req }
            'FAILED'     { throw "vRA request $RequestId FAILED: $($req.details)" }
            'ABORTED'    { throw "vRA request $RequestId ABORTED: $($req.details)" }
        }
    } while ((Get-Date) -lt $deadline)

    throw "Timed out after $TimeoutMinutes min waiting for vRA request $RequestId (last status: $($req.status))."
}

function Remove-VraVmFromDeployment {
    [CmdletBinding(SupportsShouldProcess)]
    param(
        [Parameter(Mandatory)] [string] $BaseUrl,
        [Parameter(Mandatory)] [string] $BearerToken,
        [Parameter(Mandatory)] [string] $VmName,
        [string] $Reason = "Decommission $VmName"
    )

    Write-Host "##[command] ##[FUNCTION] Remove-VraVmFromDeployment -VmName $VmName"

    $deployment = Get-VraDeployment -BaseUrl $BaseUrl -BearerToken $BearerToken -VmName $VmName
    $resources  = Get-VraDeploymentResources -BaseUrl $BaseUrl -BearerToken $BearerToken -Deployment $deployment
    $target     = Get-VraVmResources -Resources $resources -VmName $VmName

    # Last machine standing -> delete the whole deployment
    if ($target.TotalMachineCount -eq 1) {
        Write-Host "##[section] $VmName is the only machine in deployment $($deployment.id) - deleting entire deployment"
        if ($PSCmdlet.ShouldProcess($deployment.name, 'Delete deployment')) {
            $req = Remove-VraDeploymentWhole -BaseUrl $BaseUrl -BearerToken $BearerToken -DeploymentId $deployment.id -Reason $Reason
            Wait-VraRequest -BaseUrl $BaseUrl -BearerToken $BearerToken -RequestId $req.id | Out-Null
        }
        return
    }

    $machine = $target.Machine
    Write-Host "##[section] Deleting $($machine.name) ($VmName) + $($target.Volumes.Count) volume(s) from deployment of $($target.TotalMachineCount) machines"

    # --- Machine ---
    $machineActions = Get-VraResourceActions -BaseUrl $BaseUrl -BearerToken $BearerToken -DeploymentId $deployment.id -ResourceId $machine.id
    $deleteAction   = $machineActions | Where-Object { $_.valid -and $_.id -match '\.Machine\.Delete$' } | Select-Object -First 1

    if (-not $deleteAction) {
        $valid = ($machineActions | Where-Object valid | ForEach-Object id) -join ', '
        throw "No valid Delete action on $($machine.name). Valid actions: $valid. Check the day-2 policy for project $($deployment.projectId)."
    }

    if ($PSCmdlet.ShouldProcess("$VmName ($($machine.name))", $deleteAction.id)) {
        $req = Invoke-VraResourceAction -BaseUrl $BaseUrl -BearerToken $BearerToken -DeploymentId $deployment.id -ResourceId $machine.id -ActionId $deleteAction.id -Reason $Reason
        Wait-VraRequest -BaseUrl $BaseUrl -BearerToken $BearerToken -RequestId $req.id | Out-Null
    }

    # --- Volumes: non-persistent disks normally go with the VM; clean up any that remain ---
    if ($target.Volumes.Count -gt 0 -and -not $WhatIfPreference) {
        $remaining = Get-VraDeploymentResources -BaseUrl $BaseUrl -BearerToken $BearerToken -Deployment $deployment
        $volIds    = $target.Volumes | ForEach-Object id

        foreach ($vol in @($remaining | Where-Object { $volIds -contains $_.id })) {
            $volActions = Get-VraResourceActions -BaseUrl $BaseUrl -BearerToken $BearerToken -DeploymentId $deployment.id -ResourceId $vol.id
            $volDelete  = $volActions | Where-Object { $_.valid -and $_.id -match '\.Delete$' } | Select-Object -First 1

            if (-not $volDelete) {
                Write-Warning "$($vol.name) still present and has no valid Delete action - manual cleanup required."
                continue
            }

            if ($PSCmdlet.ShouldProcess($vol.name, $volDelete.id)) {
                $req = Invoke-VraResourceAction -BaseUrl $BaseUrl -BearerToken $BearerToken -DeploymentId $deployment.id -ResourceId $vol.id -ActionId $volDelete.id -Reason $Reason
                Wait-VraRequest -BaseUrl $BaseUrl -BearerToken $BearerToken -RequestId $req.id | Out-Null
            }
        }
    }

    Write-Host "##[section] $VmName removed from deployment $($deployment.id)"
}
